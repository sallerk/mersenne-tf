# ===========================================================================
#  mersenne_tf 1.3 -- machine settings.
#
#  WHAT TO WORK ON LIVES IN worktodo.txt, not here.  This file is settings for
#  this machine: which GPU, how many sieve threads, where results go.  Keeping
#  them apart means a worktodo can be swapped, or handed to another machine,
#  without disturbing anything tuned for this one.
#
#  Lines starting with # are comments.  Every value may be written as plain
#  decimal (1234567), with digit separators (1_234_567), or as a power of two
#  (2^90, 2^90-1).
#
#  Values are checked when the file is read.  A number that does not parse, or
#  one outside the range its key accepts, stops the program with the offending
#  line number rather than quietly falling back to a default.  The accepted
#  range is listed with each key below.
# ===========================================================================


# ---------------------------------------------------------------------------
#  THE JOB FILE.
#  Where to read the assignment from.  See worktodo.txt for its format.
# ---------------------------------------------------------------------------
worktodo_file = worktodo.txt


# ---------------------------------------------------------------------------
#  PRE-FACTORING (sieve).
#  Candidates q = 2kp+1 that are divisible by a prime below this bound are
#  discarded on the CPU and never reach the GPU, so no GPU time is spent on
#  numbers that are obviously composite.  Higher = more candidates removed but
#  more CPU work per candidate; the best value is where the CPU sieve and the
#  GPU finish at the same rate.
#
#  "auto" picks it from the size of the exponent and from where the sieve runs:
#  2000000 / 4000000 on the device, 500000 / 1000000 on the CPU (the lower of
#  each pair for exponents up to 32 bits).  The curve is flat, so there is little
#  to gain by hand-tuning unless you are benchmarking.
#
#  The device bracket was 120000 / 250000 up to 1.2.0, because the device sieve
#  then got rapidly more expensive with depth -- its cost scaled with the NUMBER
#  OF PRIMES rather than with the number of strikes, so past ~120000 each extra
#  prime cost more than the candidates it removed.  sieve_mark_huge removed that
#  scaling and the curve inverted: deeper is now better up to a few million.
#  Measured, p=86000009, 2^66..2^67, RTX 3070, whole bit level in seconds:
#
#     sieve_primes  120k  250k  500k  1.04M  2M    4M    8M    16M   32M
#     1.2.0         9.48  9.82  11.88 16.78  --    --    --    --    --
#     now           8.41  8.19   8.05  7.90  7.76  7.70  7.60  7.70  8.05
#  (Sieve primes that fall inside your range are tested directly, so raising
#   this can never hide a factor.)
#
#  CEILING.  A prime that strikes fewer than about one position in eight of a
#  segment does not remove enough candidates to pay for its own setup, so the
#  sieve stops at segment_size/8 and this value is limited to that.  At the
#  default segment_size = 4194304 the ceiling is 524288: anything above it has
#  no effect except the cost of building a larger prime table.  The run header
#  says so when it applies --
#      pre-sieve  : primes below 1000000, capped to 524288 by segment_size/8
#  -- and "auto" above 32 bits asks for 1000000, so it meets that ceiling.
#  Raising segment_size raises the ceiling, but measurement says that is a bad
#  trade: letting those primes through removes a few percent more candidates and
#  costs appreciably more wall time.  The ceiling is there for that reason, not
#  as something to tune around.
#
#  Range: auto, or 2 .. 1000000000, effective value limited to segment_size/8.
# ---------------------------------------------------------------------------
#  TUNING IT.  The best value depends on how fast your machine sieves relative
#  to how fast it tests, so it differs between machines -- the shipped default
#  came from the machine this was developed on and is a starting point, not a
#  universal optimum.  It also depends on the `sieve` setting below: moving the
#  sieve onto the device roughly halves it, because there the two sides compete
#  for the same GPU instead of overlapping.  To find yours, time one bit level
#  at a few values and keep the fastest; the M/s on the progress line counts
#  every candidate disposed of, so it is directly comparable between them.
#
#  Expect a flat curve near the optimum -- being off by a factor of a few costs
#  only a few percent -- and a steep penalty for sieving far too shallow, since
#  every prime you skip hands the GPU candidates it need not have tested.
#  Sieving deeper than the optimum costs less, but still costs.
#  The measurements behind the shipped default are in CHANGELOG.md.
#  Since 1.2 the M/s on the progress line counts every candidate disposed of,
#  sieved or tested, so it moves the same way as the wall time and can be
#  compared across sieve depths.  (Up to 1.1 it counted GPU arrivals only and
#  fell as the sieve improved, which made it useless for exactly this job.)
sieve_primes = 5000000


# ---------------------------------------------------------------------------
#  CHECKPOINTS.
#  The run is a loop over bit levels and, inside each, over wheel classes, so a
#  class boundary is a safe restart point.  A checkpoint is written every
#  checkpoint_seconds, at every level boundary, and on Ctrl-C (at the last class
#  that finished -- an interrupted class is re-done, never skipped), and is
#  deleted when the range completes.  Re-run with the same config to resume; a
#  checkpoint is only accepted if the exponent and both bounds still match, and
#  files written before the level-by-level scan order are rejected (they count
#  classes across the whole range, not within a level).
#    checkpoint          1 = on, 0 = off  (also true/false, yes/no, on/off)
#    checkpoint_file     auto = checkpoint_<exponent>.txt
#    checkpoint_seconds  minimum interval between writes, 0 .. 86400
# ---------------------------------------------------------------------------
checkpoint         = 1
checkpoint_file    = auto
checkpoint_seconds = 30


# ---------------------------------------------------------------------------
#  WHERE THE SIEVE RUNS.
#    gpu  pre-factoring happens on the device: the bitmap is marked, compacted
#         and consumed without the candidate list ever crossing PCIe, and the
#         CPU does nothing per candidate.  This is what 1.2 added.
#    cpu  the 1.1 pipeline: mark on the CPU, pack survivors into pinned memory,
#         upload four bytes each.  Kept as the reference implementation -- the
#         two produce bit-for-bit identical survivor sets (verified over 89.1e9
#         candidates at sieve_primes = 50000, where the segment_size/8 ceiling
#         below cannot bind and both paths therefore use the same primes).
#
#  In testing gpu is the faster of the two, and because it uses almost no CPU
#  per candidate it is far less affected by whatever else the machine is doing.
#  The sieve kernels were the whole of the remaining gap against mfakto and are
#  now within ~13% of it; the attribution, and the two changes that closed it,
#  are in CHANGELOG.md.
#
#  Note the ceiling documented under sieve_primes is a CPU-cost heuristic and
#  applies to `cpu` only; the device path applies every prime on every segment,
#  which is why its survivor counts are very slightly lower at large
#  sieve_primes.  It removes more composites, never fewer.
# ---------------------------------------------------------------------------
sieve = gpu


# ---------------------------------------------------------------------------
#  CANDIDATES PER WORK ITEM.
#    auto  2 where a vectorised kernel exists (the 72- and 96-bit paths)
#    1     one candidate per thread
#    2     two independent candidates interleaved in one thread
#
#  A single candidate is one serial dependency chain -- every squaring needs the
#  previous result -- so a 1-wide kernel waits on latency rather than running
#  short of work.  Measured worth 2.8% when the GPU is the constraint and 0.6%
#  when it is not; set it to 1 only to compare.
# ---------------------------------------------------------------------------
vector = auto


# ---------------------------------------------------------------------------
#  ARITHMETIC WIDTH.
#  auto picks the narrowest kernel that is exact for the bit level being
#  scanned, and re-picks at every level:
#     q < 2^70   three 24-bit limbs, lazy reduction   (fastest)
#     q < 2^82   three 28-bit limbs, lazy reduction
#     q < 2^96   three 32-bit limbs
#     otherwise  two 64-bit limbs
#  (three 24-bit limbs, eager, still exists for q < 2^72 but auto no longer
#   selects it -- the 28-bit path is 1-wide-vs-2-wide faster over that strip.)
#  Narrower is faster because the cost is the accumulation around the multiplies
#  rather than the multiplies themselves -- sizing the limbs to the candidate
#  measures 1.46x over the 96-bit path, and skipping the per-squaring reduction
#  1.78x.  Force 64/72/84/96/128 only to compare them; a forced width that cannot
#  hold the candidates is ignored rather than silently used.
#
#  The 28-bit path is new.  The 24-bit one is fast not because 24 is special but
#  because a 24x24 product is 48 bits, so a 64-bit column holds several of them
#  plus carries without normalising -- and that holds up to 30-bit limbs, not 24.
#  Everything from 2^72 to 2^96 used to fall back to three 32-bit limbs, whose
#  columns do not fit; three 28-bit limbs reach 2^82 with the same fifteen
#  products and the cheap accumulation, worth 21% on the kernel and 1.18x on a
#  whole run at 2^76 -- which is the range GIMPS trial factoring actually uses.
# ---------------------------------------------------------------------------
arithmetic = auto


# ---------------------------------------------------------------------------
#  TUNING.  Leave these alone unless you are benchmarking.
#    segment_size  candidates per sieve segment / GPU batch (power of two).
#                  2^21..2^23 all measure the same; below that per-batch overhead
#                  shows up.  Do NOT raise it far: the pinned staging buffers are
#                  4 bytes per candidate each, so 2^24 needs 64 MB per buffer,
#                  the pool falls back to its 3-buffer minimum and the pipeline
#                  starves -- measured at well under half the throughput of
#                  2^21..2^23.
#                  Those figures are for sieve = cpu.  The device sieve wants a
#                  much larger segment, because it launches several kernels per
#                  segment and small segments pay that overhead over and over,
#                  while very large ones spill the index buffers out of GPU
#                  memory.  2^24 measured best; see CHANGELOG.md.
#                  It also sets the CPU sieve's ceiling: no prime above
#                  segment_size/8 is applied there, so at 2^24 that ceiling is
#                  2097152 -- above any sensible sieve_primes, so it no longer
#                  binds.  (At the old 2^22 default it was 524288 and did.)
#                  Range 4096 .. 2^28.
#    threads       CPU sieve threads, 0 = auto (cores - 1).  Range 0 .. 4096.
#    workgroup     GPU work-group size, 0 = ask the driver.  Measured flat from
#                  64 to 1024 in testing; only 32 was slower.  Range 0 .. 65536.
#    gpu_slots     batches in flight on the device (2..8).  2 costs ~5%; 3, 4
#                  and 6 measured within noise of each other.
#    platform      OpenCL platform index, -1 = auto     (--list-devices)
#    device        OpenCL device index on that platform, 0 or above
# ---------------------------------------------------------------------------
segment_size = 16777216
threads      = 0
workgroup    = 0
gpu_slots    = 3
platform     = -1
device       = 0


# ---------------------------------------------------------------------------
#  OUTPUT.
#    stop_on_factor  1 = stop as soon as a factor is found, 0 = scan it all
#    results_file    GIMPS manual-submission lines, and nothing else, so the
#                    file can be pasted straight into
#                    https://www.mersenne.org/manual_result/ :
#                        M9147253 has a factor: 313603386094415369 [TF:64:65:...]
#                        no factor for M9147253 from 2^64 to 2^65 [...]
#                    A "no factor" line is written as each bit level clears, so
#                    a run stopped part way still submits what it finished.  A
#                    level the range only partly covers is NOT reported -- that
#                    claim belongs to whoever finishes it.
#    log_file        the human record: one line per run including interrupted
#                    ones, with status, candidate counts and timings.  This is
#                    the file to read when you want to know what happened; it is
#                    kept separate so results.txt stays machine-parseable.
#    pause_on_exit   auto   = wait for Enter only when double-clicked (default)
#                    always = always wait for Enter before closing
#                    never  = never wait (for scripts / batch jobs)
#
#  Factors are printed the instant the GPU finds them, not at the end of the
#  run, so you can leave a long job going and watch the window.
# ---------------------------------------------------------------------------
stop_on_factor = 0
results_file   = results.txt
log_file       = runlog.txt
pause_on_exit  = auto
